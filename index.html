  <script>

    const params = new URLSearchParams(window.location.search);

    const name = params.get("name");

    const email = params.get("email");

    const ref = params.get("ref");


    if (name) {

      document.getElementById("message").textContent =

        `Thank you, ${name}. We have received your submission and will follow up shortly.`;

    }


    if (email || ref) {

      const detailsBox = document.getElementById("details");

      detailsBox.hidden = false;


      const refValue = ref || `REF-${Math.random().toString(36).substring(2, 8).toUpperCase()}`;

      document.getElementById("ref-id").textContent = refValue;

      document.getElementById("user-email").textContent = email || "Confirmed";


      const copyBtn = document.getElementById("copy-btn");

      copyBtn.hidden = false;

      copyBtn.addEventListener("click", async () => {

        try {

          await navigator.clipboard.writeText(refValue);

          copyBtn.textContent = "Copied!";

          setTimeout(() => (copyBtn.textContent = "Copy Reference"), 2000);

        } catch {

          copyBtn.textContent = refValue;

        }

      });

    }
